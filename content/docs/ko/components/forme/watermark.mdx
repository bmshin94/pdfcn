| `text`     | `string`                                                                   | -      |
| `opacity`  | `number`                                                                   | -      |
| `fontSize` | `number`                                                                   | -      |
| `color`    | `string`                                                                   | -      |
| `angle`    | `number`                                                                   | -      |
| `position` | `'center' \| 'top-left' \| 'top-right' \| 'bottom-left' \| 'bottom-right'` | -      |
| `fixed`    | `boolean`                                                                  | true   |
| `children` | `never`                                                                    | -      |
