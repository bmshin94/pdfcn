## 使い方

```tsx
import {
  Table,
  TableBody,
  TableCell,
  TableHeader,
  TableRow,
  Text,
} from "@/components/pdf/table";
```

```tsx
<Table>
  <TableHeader>
    <TableRow>
      <TableCell>
        <Text>Item</Text>
      </TableCell>
    </TableRow>
  </TableHeader>
</Table>
```

### バリアント

```tsx
<Table variant="striped" zebraStripe>
  {/* header, body, rows, and cells */}
</Table>
```

## APIリファレンス

### Table

| プロパティ    | 型                                                                                          | デフォルト |
| ------------- | ------------------------------------------------------------------------------------------- | ---------- |
| `variant`     | `'line' \| 'grid' \| 'minimal' \| 'striped' \| 'compact' \| 'bordered' \| 'primary-header'` | `'line'`   |
| `zebraStripe` | `boolean`                                                                                   | `false`    |
| `noWrap`      | `boolean`                                                                                   | `false`    |
| `children`    | `ReactNode`                                                                                 | -          |
| `style`       | `Style`                                                                                     | -          |

### TableRow

| プロパティ | 型                                                                                          | デフォルト |
| ---------- | ------------------------------------------------------------------------------------------- | ---------- |
| `header`   | `boolean`                                                                                   | `false`    |
| `footer`   | `boolean`                                                                                   | `false`    |
| `stripe`   | `boolean`                                                                                   | `false`    |
| `variant`  | `'line' \| 'grid' \| 'minimal' \| 'striped' \| 'compact' \| 'bordered' \| 'primary-header'` | inherited  |

### TableCell

| プロパティ | 型                                                                                          | デフォルト |
| ---------- | ------------------------------------------------------------------------------------------- | ---------- |
| `header`   | `boolean`                                                                                   | `false`    |
| `footer`   | `boolean`                                                                                   | `false`    |
| `align`    | `'left' \| 'center' \| 'right'`                                                             | `'left'`   |
| `width`    | `string \| number`                                                                          | -          |
| `variant`  | `'line' \| 'grid' \| 'minimal' \| 'striped' \| 'compact' \| 'bordered' \| 'primary-header'` | inherited  |
